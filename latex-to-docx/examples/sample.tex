\section{Introduction}
This document demonstrates converting \LaTeX{} source --- \textbf{text},
\emph{equations}, and images --- into a Microsoft Word (\texttt{.docx})
file. Inline math such as $E = mc^2$ or $\nu = \mu/\rho$ becomes a
native, editable Word equation rather than a picture.

\section{Equations}
The continuity equation for incompressible flow:
\begin{equation}
  \nabla \cdot \mathbf{u} = 0
  \label{eq:continuity}
\end{equation}

The Navier--Stokes momentum equation:
\begin{equation}
  \rho\left(\frac{\partial \mathbf{u}}{\partial t}
    + \mathbf{u}\cdot\nabla\mathbf{u}\right)
  = -\nabla p + \mu\nabla^{2}\mathbf{u} + \rho\mathbf{g}
\end{equation}

Aligned equations also work:
\begin{align}
  \int_{0}^{\infty} e^{-x^{2}}\,dx &= \frac{\sqrt{\pi}}{2} \\
  \sum_{n=1}^{\infty} \frac{1}{n^{2}} &= \frac{\pi^{2}}{6}
\end{align}
